CMAKE_MINIMUM_REQUIRED(VERSION 3.16)

PROJECT(ai_pipe)

SET(CMAKE_POSITION_INDEPENDENT_CODE ON)

LOAD_OPENCV()
LOAD_COMMON_UTILS()
LOAD_LOGGER()
LOAD_AI_CORE()

FILE(GLOB_RECURSE CURRENT_DIR_HEAD 
	${CMAKE_CURRENT_SOURCE_DIR}/*.hpp 
	${CMAKE_CURRENT_SOURCE_DIR}/types/*.hpp
	${CMAKE_CURRENT_SOURCE_DIR}/tracker/*.hpp
)

FILE(GLOB_RECURSE CURRENT_DIR_SRCS 
		${CMAKE_CURRENT_SOURCE_DIR}/*.cpp 
		${CMAKE_CURRENT_SOURCE_DIR}/tracker/*.cpp
)

SET(DEPENDENCY_INCLUDES
	${PROJECT_INCLUDE_DIR} 
	${PROJECT_INCLUDE_DIR}/ai_pipe
	${OpenCV_INCLUDE_DIRS} 
)

SET(DEPENDENCY_LIBS
	ai_core::ai_core
	common_utils::common_utils
	logger::logger
	${OpenCV_LIBS}
)

# Build Shared Library
ADD_LIBRARY(ai_pipe ${CURRENT_DIR_SRCS})

ADD_LIBRARY(ai_pipe::ai_pipe ALIAS ai_pipe)

TARGET_INCLUDE_DIRECTORIES(ai_pipe 
	PUBLIC
		$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
		$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/types>
	PRIVATE 
		${DEPENDENCY_INCLUDES}
)

TARGET_LINK_LIBRARIES(ai_pipe PRIVATE ${DEPENDENCY_LIBS})
